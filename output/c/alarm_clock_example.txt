StateMachine M {
	Start State IDLE {
		ON ALARM_BTN_PRESS GO ALARM_ARMED;
	}

	State ALARM_ARMED {
		ON ALARM_BTN_PRESS GO IDLE;
		ON ALARM_TIMER_EXPIRED GO ALARM_RINGING;
	}

	State ALARM_RINGING {
		ON ALARM_BTN_PRESS GO IDLE;
		ON SNOOZE_BTN_PRESS GO SNOOZE;
	}

	State SNOOZE {
		ON SNOOZE_TIME_EXPIRED GO ALARM_RINGING;
		ON ALARM_BTN_PRESS GO IDLE;
	}
}
